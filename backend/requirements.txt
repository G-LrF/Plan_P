# 后端依赖清单（Python 包）
#
# 【怎么用】（在 backend 目录下）
#     ..\.venv\Scripts\python.exe -m pip install -r requirements.txt
#
# 【为什么必须有这个文件】
#   你现在的 .venv 装了什么，只有你这台机器知道。换一台电脑、
#   或者将来放进 Docker 镜像里，就必须有一份清单能"重建出一模一样的环境"。
#   这是可复现构建的基础——Docker/K8s 部署时读的就是它。
#
# 【为什么用 >= 而不是 == 精确锁版本】
#   精确锁版本（==）在生产环境更安全（避免自动升级引入意外行为），
#   但现在处于学习阶段，用 >= 更容易装上、不容易因为版本找不到而失败。
#   将来进企业或在 Docker 里构建镜像时，建议改用 pip freeze 生成精确版本。
#
# 以下版本是照你当前 .venv 里实际安装的版本记录的，不是我凭空写的。

fastapi>=0.143.0
uvicorn>=0.54.0
# pydantic 是 fastapi 的依赖，会连带装上，但显式写出来更清楚：
# 我们的 schema.py 直接 import 了它，属于"直接依赖"，不该藏在间接依赖里。
pydantic>=2.13.0

# ---------- 数据库（阶段 1 新增）----------
# SQLAlchemy：Python 最主流的 ORM，负责"Python 对象 ↔ 数据库表"的映射
sqlalchemy>=2.0.0

# PyMySQL：MySQL 的纯 Python 驱动
# 【为什么不用 mysqlclient】那个需要编译 C 扩展，Windows 上装起来很折腾。
# PyMySQL 是纯 Python 实现，装起来最省事，小项目性能完全够用。
pymysql>=1.1.0

# pydantic-settings：从 .env 读配置
# 【为什么需要它】让数据库密码不写在代码里。
# 代码要提交到 Git，.env 不提交 —— 这是避免密码泄露的标准做法。
pydantic-settings>=2.0.0
